  match l with
  | [] => EmptyString
  | c :: cs => String c (string_of_list_ascii cs)
  end.

(* Schema based on Cedar's sample apps*)
Definition schema : Schema :=
  (* Entity types *)
  let team := MkName "Team" [] in
  let teamDef := MkEntitySchemaEntry [team] [] in
  let user := MkName "User" [] in
  let userDef := MkEntitySchemaEntry [team]
    [("manager", true, entityType user)] in
  let lst := MkName "List" [] in
  let lstDef := MkEntitySchemaEntry []
    [("owner", true, entityType user);
     ("readers", true, entityType team);
     ("editors", true, entityType team);
     ("age", true, intType);
     ("description", true, stringType)] in
  let app := MkName "Application" [] in
  let appDef := MkEntitySchemaEntry [] [] in
  let action := MkName "Action" [] in
  let actionDef := MkEntitySchemaEntry [action] [] in
  let ets := [(user, userDef); (team, teamDef); (lst, lstDef); (app, appDef); (action, actionDef)] in
  
  (* Action definitions *)
  let getAct := MkActionSchemaEntry [user] [lst] [] in
  let createAct := MkActionSchemaEntry [user] [app] [] in
  let updateAct := MkActionSchemaEntry [user] [lst] [] in
  let acts := [(MkEntityUID action "getList", getAct);
               (MkEntityUID action "createList", createAct);
               (MkEntityUID action "updateList", updateAct)] in
  MkSchema ets acts.

(* ------------------------------------------------------------------ *)
(* Schema serializer producing valid cedarschema syntax.               *)
(*  show (MkSchema ...) has three bugs that make Cedar reject it:      *)
(*  1. actions use "Type::\"id\"" instead of just "\"id\""             *)
(*  2. appliesTo parts are separated by ";" instead of ","             *)
(*  3. the required context clause must include generated context attrs *)
(*  4. optional attrs are "name:? T" instead of "name?: T"             *)
(*  5. intType serialises as "Int", Cedar schema needs "Long"          *)
(* ------------------------------------------------------------------ *)

Fixpoint typeToSchemaStr (t : CedarType) : string :=
  match t with
  | intType    => "Long"
  | stringType => "String"
  | boolType _ => "Bool"
  | entityType n => stringOfEntityNameFull n
  | setType t' => "Set<" ++ typeToSchemaStr t' ++ ">"
  | recordTypeNil => "{}"
  | recordTypeCons fn o t' recordTypeNil =>
      "{ " ++ quoteIfNeeded fn ++ (if o then "?: " else ": ") ++ typeToSchemaStr t' ++ " }"
  | recordTypeCons fn o t' rest =>
      "{ " ++ quoteIfNeeded fn ++ (if o then "?: " else ": ") ++ typeToSchemaStr t' ++
      ", " ++ recordTypeToSchemaStr rest ++ " }"
  end

with recordTypeToSchemaStr (t : CedarType) : string :=
  match t with
  | recordTypeNil => ""
  | recordTypeCons fn o t' recordTypeNil =>
      quoteIfNeeded fn ++ (if o then "?: " else ": ") ++ typeToSchemaStr t'
  | recordTypeCons fn o t' rest =>
      quoteIfNeeded fn ++ (if o then "?: " else ": ") ++ typeToSchemaStr t' ++
      ", " ++ recordTypeToSchemaStr rest
  | _ => ""
  end.

Fixpoint attrsToSchemaStr (attrs : list (string * bool * CedarType)) : string :=
  match attrs with
  | nil => ""
  | cons (s, o, t) nil =>
      quoteIfNeeded s ++ (if o then "?: " else ": ") ++ typeToSchemaStr t
  | cons (s, o, t) rest =>
      quoteIfNeeded s ++ (if o then "?: " else ": ") ++ typeToSchemaStr t
      ++ ", " ++ attrsToSchemaStr rest
  end.

Fixpoint etsToSchemaStr (ets : list (EntityName * EntitySchemaEntry)) : string :=
  match ets with
  | nil => ""
  | cons (n, MkEntitySchemaEntry ancs attrs) rest =>
      "entity " ++ stringOfEntityNameFull n ++
      (match ancs with
       | nil => ""
       | _   => " in [" ++ stringOfEntityNameList ancs ++ "]"
       end) ++
      (match attrs with
       | nil => ""
       | _   => " { " ++ attrsToSchemaStr attrs ++ " }"
       end) ++
      "; " ++ etsToSchemaStr rest
  end.

Fixpoint actsToSchemaStrAcc
    (seen_ids : list string)
    (acts : list (EntityUID * ActionSchemaEntry)) : string :=
  match acts with
  | nil => ""
  | cons (uid, MkActionSchemaEntry ps rs ctx) rest =>
      let id := match uid with MkEntityUID _ i => i end in
      if List.existsb (String.eqb id) seen_ids then
        actsToSchemaStrAcc seen_ids rest
      else
        "action " ++ """" ++ id ++ """" ++
        " appliesTo { principal: [" ++ stringOfEntityNameList ps ++
        "], resource: ["             ++ stringOfEntityNameList rs ++
        "], context: {"              ++ attrsToSchemaStr ctx ++
        "} }; "                      ++ actsToSchemaStrAcc (id :: seen_ids) rest
  end.

Definition actsToSchemaStr (acts : list (EntityUID * ActionSchemaEntry)) : string :=
  actsToSchemaStrAcc [] acts.

Definition schemaToStr (s : Schema) : string :=
  match s with
  | MkSchema ets acts => etsToSchemaStr ets ++ actsToSchemaStr acts
  end.

Definition actionUIDToCedarStr (u : EntityUID) : string :=
  match u with
  | MkEntityUID _ id => "Action::" ++ """" ++ id ++ """"
  end.

(* Pull the context info out of an environment and call the Rust checker.
   Wrapping the match in a Definition keeps the theorem conclusion match-free
   so QuickChick can process it. *)
Definition no_type_error_with_env
    (sc : Schema) (e : Environment) (expr : CedarExpr * PathSet) : bool :=
  match e with
  | MkEnvironment _ (MkRequest P action_uid R _) =>
    cedar_no_type_error_with_ctx_coq
      (list_ascii_of_string' (schemaToStr sc))
      (list_ascii_of_string' (stringOfEntityNameFull P ++ "::" ++ """" ++ "e1" ++ """"))
      (list_ascii_of_string' (actionUIDToCedarStr action_uid))
      (list_ascii_of_string' (stringOfEntityNameFull R ++ "::" ++ """" ++ "e2" ++ """"))
      (list_ascii_of_string' (stringOfExpr (fst expr)))
  end.

Theorem prop_no_type_error :
  forall ets acts rs e es ty expr,
    WfSchema [MkName "EntityType1" []; MkName "EntityType2" []] (MkSchema ets acts) ->
    ActionSchemaToRequestTypes acts [] rs ->
    SchemaToEnvironments (MkSchema ets acts) rs (cons e es) ->
    WfCedarType [MkName "EntityType1" []; MkName "EntityType2" []] ty ->
    HasType (somepaths []) e expr ty ->
    no_type_error_with_env (MkSchema ets acts) e expr = true.
  quickchick 2.
Admitted.
